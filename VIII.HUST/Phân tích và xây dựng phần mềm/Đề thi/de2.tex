\noindent\textbf{Phần 1. (4 điểm) Trắc nghiệm, viết câu trả lời vào trong bài làm}

\cau{1}{Vì sao Analysis và Design được nói là ``không phải top-down cũng không phải bottom-up''?}
\begin{pa}
\A {\bfseries Vì use case đi vào từ giữa, định nghĩa một mức trung gian (middle level); từ đó có thể đi lên (subsystem) hoặc đi xuống (design class)}
\B Vì luôn bắt đầu từ design class rồi đi lên subsystem
\C Vì luôn bắt đầu từ subsystem rồi đi xuống design class
\D Vì analysis class được định nghĩa hoàn toàn theo hướng top-down
\end{pa}

\cau{2}{Khi chuyển đổi từ Analysis Classes sang Design Elements, mối quan hệ ánh xạ giữa chúng có đặc điểm gì? Chọn các phát biểu đúng.}
\begin{pa}
\A Một analysis class luôn ánh xạ đúng một design class tương ứng
\B {\bfseries Một analysis class có thể ánh xạ tới nhiều design elements khác nhau}
\C {\bfseries Nhiều analysis classes có thể cùng ánh xạ tới một design element}
\D Một analysis class loại \texttt{<<control>>} chỉ có thể ánh xạ tới một subsystem duy nhất
\E Một analysis class loại \texttt{<<entity>>} không thể trở thành một subsystem
\end{pa}

\cau{3}{Nhóm phát triển hệ thống quản lý bệnh viện xác định được analysis class \texttt{<<control>> TreatmentCoordinator} có các trách nhiệm sau: (1) điều phối quy trình khám chữa bệnh, (2) tính toán chi phí điều trị, (3) giao tiếp với hệ thống bảo hiểm bên ngoài, (4) lập lịch phòng mổ. Khi chuyển sang thiết kế, nhóm nên xử lý class này như thế nào? Chọn các phương án đúng.}
\begin{pa}
\A Giữ nguyên thành một design class vì đây là control class điều phối trung tâm
\B Gộp với class khác vì trách nhiệm của lớp này quá đơn giản
\C Chuyển toàn bộ thành một subsystem vì có giao tiếp với hệ thống ngoài
\D {\bfseries Tách phần giao tiếp với hệ thống bảo hiểm thành một interface riêng biệt}
\E {\bfseries Tách thành package chứa nhiều classes chuyên biệt để quản lý độ phức tạp}
\end{pa}

\cau{4}{Những phát biểu nào sau đây là đúng về vai trò của Subsystem so với Package?}
\begin{pa}
\A Cả subsystem và package đều cung cấp hành vi (behavior) cho hệ thống
\B {\bfseries Subsystem đóng gói hoàn toàn nội dung bên trong, trong khi package thì không}
\C {\bfseries Package dễ dàng thay thế hơn subsystem trong kiến trúc hệ thống}
\D {\bfseries Subsystem nâng cao mức độ trừu tượng hóa (level of abstraction) của hệ thống}
\E Subsystem và package đều có thể dễ dàng thay thế nhau mà không ảnh hưởng đến client
\end{pa}

\cau{10}{Điều gì phân biệt quan hệ phụ thuộc (dependency/identifying)\\ với quan hệ độc lập (independency/non-identifying) giữa quan hệ thực thể cha và con?}

\cau{12}{Những phát biểu nào đúng về thiết kế lớp boundary?}

\cau{14}{Khi thiết kế chữ ký operation, những cân nhắc nào là đúng?}
\begin{pa}
\A {\bfseries Càng ít tham số càng tốt, giúp tăng tính dễ hiểu và dễ bảo trì.}
  \B {\bfseries Nên truyền nguyên cả đối tượng (ví dụ Student) thay vì truyền từng mảnh dữ liệu rời rạc (ví dụ StudentID).}
  \C {\bfseries Nếu tham số không bao giờ bị thay đổi bởi operation, nên thiết kế sao cho không thể thay đổi nó.}
  \D Càng nhiều tham số thì các đối tượng càng ít bị coupling về mặt khái niệm.
  \E {\bfseries Truyền by value nghĩa là truyền một con trỏ/tham chiếu tới đối tượng.}
\F Không quan tâm tham số là optional hay có giá trị mặc định.
\end{pa}

\cau{15}{Những phát biểu đúng về navigability trong Class Design?}

  \B Navigability được chốt (addressed) trong Use-Case Analysis chứ không phải trong Class Design.
  \C {\bfseries Mặc định association là hai chiều (bi-directional); khi không có mũi tên, giả định điều hướng được cả hai hướng.}

\cau{16}{Những phát biểu đúng về Generalization / Inheritance?}
\begin{pa}
\A Trong generalization, object của superclass có thể thay thế tùy ý cho object của subclass.
\B {\bfseries Generalization là quan hệ ``is a kind of'' và có tính bắc cầu (transitive).}
\C {\bfseries Subclass có thể dùng nơi superclass được dùng, nhưng không tùy ý ngược lại.}
\D {\bfseries ``Generalization'' là tên quan hệ, còn ``Inheritance'' là cơ chế hiện thực quan hệ đó.}
\E Multiple inheritance nghĩa là một superclass có nhiều subclass.
\end{pa}

\cau{17}{Những phát biểu nào đúng về so sánh quan hệ \texttt{<<include>>} và \texttt{<<extend>>}?}
\begin{pa}
\A {\bfseries Trong \texttt{<<include>>}, use case được include luôn luôn được thực thi.}
  \B Trong \texttt{<<extend>>}, use case mở rộng luôn luôn được thực thi.
  \C {\bfseries Base use case trong quan hệ \texttt{<<include>>} sẽ không đầy đủ nếu thiếu use case được include.}
  \D {\bfseries Base use case trong quan hệ \texttt{<<extend>>} vẫn hoàn chỉnh khi không có use case mở rộng.}
  \E Chiều phụ thuộc của \texttt{<<include>>} và \texttt{<<extend>>} là giống nhau.
  \F \texttt{<<extend>>} được dùng để tách hành vi khác nhau cho nhiều use case.
\end{pa}

\noindent\textbf{Phần 2.} Bạn cần thiết kế và xây dựng phần mềm đặt vé xem phim trực tuyến cho một hệ thống rạp chiếu phim có nhiều cụm rạp tại các thành phố khác nhau. Mỗi cụm rạp sở hữu nhiều phòng chiếu với sức chứa khác nhau: 60, 120 và 250 ghế.
\begin{itemize}[leftmargin=1.05cm]
\item Mỗi phòng chiếu được định danh bằng mã phòng và số hiệu cụm rạp để quản lý dễ dàng. Các suất chiếu được vận hành theo lịch chiếu cố định trong ngày, mỗi phim được xếp vào các khung giờ xác định trước. Mỗi suất chiếu có một phim được công chiếu và một nhân viên vận hành phòng được phân công. Với các phim dài, hệ thống có thể bố trí thời gian giải lao giữa suất nhằm bảo đảm trải nghiệm cho khán giả.
\item Hệ thống cung cấp hai loại suất chiếu để đáp ứng nhu cầu đa dạng của khán giả:
  \begin{itemize}[leftmargin=0.75cm,itemsep=0pt,topsep=0pt,parsep=0pt]
  \item \textit{Suất chiếu thường}: Giá vé tiêu chuẩn, chiếu trên các phòng 2D thông thường, cho phép đặt mọi hạng ghế.
  \item \textit{Suất chiếu đặc biệt (3D/IMAX)}: Giá vé cao hơn, chỉ chiếu trên các phòng kỹ thuật cao, bắt buộc đặt theo khu vực ghế quy định, nhằm nâng cao chất lượng hình ảnh và âm thanh.
  \end{itemize}
\item Khách hàng có thể đăng ký tài khoản thành viên trên trang web của hệ thống để dễ dàng thực hiện đặt vé. Sau khi đặt vé, khách hàng có thể chọn phương thức thanh toán: Thanh toán trước bằng thẻ tín dụng qua hệ thống liên ngân hàng; hoặc Thanh toán sau: trả tiền trực tiếp tại quầy trước giờ chiếu.
\item Nếu khách hàng đã thanh toán trước, họ có thể hủy vé và được hoàn lại 80\% giá trị vé. Tuy nhiên, trong vòng 2 giờ trước suất chiếu, việc hủy vé không được chấp nhận. Khi một ghế bị hủy, hệ thống tự động cập nhật ghế để có thể được bán lại cho khách hàng khác. Đối với các vé chưa thanh toán, nếu khách hàng không có mặt tại quầy ít nhất 30 phút trước giờ chiếu, hệ thống có quyền hủy chỗ để nhường cho khách hàng khác.
\end{itemize}

\noindent\textbf{Trả lời các câu hỏi sau:}

\noindent a) (3 điểm) Xây dựng biểu đồ lớp (thiết kế) cho use case \textbf{``Đặt vé xem phim''} được mô tả như trên. Lưu ý: phần thiết kế của bạn không bao gồm use case được gói đến \textbf{``Thanh toán trực tuyến''}.

\par\nobreak\vspace*{0pt}
{\itshape\nobreak Luồng sự kiện chính:}
\begin{enumerate}[leftmargin=1.05cm]
\item Khách hàng truy cập tính năng đặt vé trên trang web hoặc ứng dụng của hệ thống.
\item Hệ thống hiển thị danh sách các suất chiếu của ngày hiện tại. Nếu suất chiếu nào đã bán hết ghế hoặc đã bắt đầu thì không cho phép đặt vé.
\item Khách hàng chọn cụm rạp, tên phim và ngày chiếu, sau đó gửi yêu cầu tìm kiếm.
\item Hệ thống tìm kiếm và hiển thị danh sách các suất chiếu còn ghế trống trong ngày đã chọn, kèm theo: giờ chiếu cụ thể (giờ:phút), loại phòng (60/120/250 ghế), giá vé, loại suất chiếu (thường / đặc biệt).
\item Khách hàng chọn suất chiếu mong muốn.
\item Hệ thống hiển thị thông tin chi tiết về suất chiếu, bao gồm: mã phòng, số hiệu cụm rạp, thông tin nhân viên vận hành, thời lượng và giờ kết thúc dự kiến; nếu là suất chiếu đặc biệt thì hiển thị sơ đồ khu vực ghế bắt buộc.
\item Khách hàng chọn số lượng và vị trí ghế muốn đặt, nhập thông tin liên hệ (họ tên, số điện thoại, email) và xác nhận đặt vé.
\item Hệ thống tính toán và hiển thị thông tin chi tiết đặt vé: thông tin suất chiếu; số lượng ghế, hàng ghế, đơn giá, tổng tiền cần thanh toán; thông tin liên hệ khách hàng.
\item Khách hàng chọn hình thức thanh toán: thanh toán trước qua thẻ tín dụng/ngân hàng (kết nối cổng thanh toán) hoặc thanh toán sau (tại quầy).
\item Nếu khách hàng chọn thanh toán trước, hệ thống gọi use case \textbf{``Thanh toán trực tuyến''} để xử lý giao dịch.
\item Sau khi hoàn tất đặt vé, hệ thống gửi thông báo xác nhận và hiển thị thông tin đặt vé thành công.
\item Nếu khách hàng đã thanh toán trước, hệ thống cung cấp vé điện tử (mã đặt chỗ) qua email hoặc SMS.
\end{enumerate}
\par\nobreak\vspace{0pt}
{\itshape\nobreak Luồng sự kiện thay thế:}
\begin{itemize}[leftmargin=1.05cm]
\item Nếu khách hàng nhập sai thông tin, hệ thống hiển thị thông báo lỗi và yêu cầu nhập lại.
\item Nếu không có suất chiếu nào phù hợp, hệ thống thông báo và đề xuất ngày/suất khác nếu có.
\item Nếu thanh toán trực tuyến thất bại, hệ thống yêu cầu khách hàng thử lại hoặc chọn phương thức khác.
\end{itemize}

\textbf{b) Mã nguồn Java cho phương thức hủy vé.}

\begin{verbatim}
public double huyDatVe(LocalDateTime thoiDiemHienTai) {
    if (thoiDiemHienTai == null || gioSuatChieu == null || ghe == null
            || trangThai != TrangThai.DA_DAT) {
        return 0.0;
    }

    if (daThanhToanTruoc) {
        // Cho phép hủy đúng tại T - 2 giờ; sau mốc này thì từ chối.
        if (thoiDiemHienTai.isAfter(gioSuatChieu.minusHours(2))) {
            return 0.0;
        }
        trangThai = TrangThai.DA_HUY;
        ghe.giaiPhong();
        return tongTien * 0.80;
    }

    // khachDaDenQuay là cờ được cập nhật khi quầy xác nhận khách đã đến.
    if (khachDaDenQuay
            || thoiDiemHienTai.isBefore(gioSuatChieu.minusMinutes(30))) {
        return 0.0;
    }
    trangThai = TrangThai.DA_HUY;
    ghe.giaiPhong();
    return 0.0;
}
\end{verbatim}

Vé đã thanh toán được hoàn 80\% khi hủy không muộn hơn mốc hai giờ; sau mốc đó thì không được hủy. Với vé chưa thanh toán, bổ sung trường \texttt{boolean khachDaDenQuay}; khi đến mốc 30 phút trước giờ chiếu, hệ thống chỉ hủy chỗ nếu cờ này vẫn là \texttt{false}. Ghế được giải phóng và số tiền hoàn là 0. Phương thức giữ nguyên trạng thái nếu vé đã hủy hoặc đầu vào thời gian không hợp lệ.

\textbf{c) Đánh giá SOLID.}
\begin{itemize}[leftmargin=1.05cm,itemsep=1pt]
  \item Phòng 40/180 ghế: tránh if/else theo danh sách cố định; capacity và seat map thuộc cấu hình ScreeningRoom. Đây là cách tuân thủ OCP và SRP.
  \item Gold Class/Beanbag: biểu diễn RoomType/SeatType hoặc abstraction về đặc tính phòng; không ép mọi phòng phải có hành vi/thuộc tính không phù hợp. Các implementation phải thay thế được abstraction (LSP).
  \item Giá theo phòng: đưa quy tắc vào PricingPolicy/Strategy, chẳng hạn StandardPricing và PremiumRoomPricing. ReservationService chỉ phụ thuộc PricingService, không tự biết bảng giá (SRP/OCP/DIP).
  \item Momo/ZaloPay/Apple Pay: tạo PaymentGateway interface và adapter cho từng provider. Inject abstraction vào PaymentService; thêm provider mới không sửa controller hay logic đặt vé (DIP/OCP).
\end{itemize}

\textbf{Kết luận:} một controller mỏng, entity có trách nhiệm miền vừa phải và các service chuyên trách tạo ra functional cohesion cao hơn; các interface ngăn phụ thuộc trực tiếp vào chi tiết triển khai và giữ thiết kế mở cho mở rộng.
